\section{Spectral Geometry Foundations}

\subsection{Laplace--Beltrami Operator}
\begin{definition}
Let $(\M, g)$ be a smooth Riemannian manifold.  
The \emph{Laplace--Beltrami operator} acting on smooth functions
$f \in C^\infty(\M)$ is defined by
\[
\Delta f = \mathrm{div}(\nabla f).
\]
\end{definition}

\subsection{Eigenfunctions and Eigenvalues}
\begin{definition}
A function $\phi \in C^\infty(\M)$ is a \emph{Laplace--Beltrami eigenfunction}
with eigenvalue $\lambda \in \mathbb{R}$ if
\[
\Delta \phi = \lambda \phi.
\]
\end{definition}

\subsection{Spectral Gap}
\begin{definition}
The \emph{spectral gap} of $\M$ is the difference between the first two
distinct eigenvalues of the Laplace--Beltrami operator:
\[
\gamma = \lambda_1 - \lambda_0.
\]
\end{definition}

\subsection{Heat Kernel}
\begin{definition}
The \emph{heat kernel} $K_t(x,y)$ is the fundamental solution to the heat
equation
\[
\frac{\partial u}{\partial t} = \Delta u,
\]
satisfying $u(x,0) = f(x)$ and
\[
u(x,t) = \int_{\M} K_t(x,y) f(y) \, dy.
\]
\end{definition}
