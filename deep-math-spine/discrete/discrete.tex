\section{Discrete Deterministic Systems}

\subsection{Deterministic Finite Automata (DFA)}
\begin{definition}
A \emph{deterministic finite automaton} is a tuple
\[
A = (Q, \Sigma, \delta, q_0, F)
\]
where $Q$ is a finite set of states, $\Sigma$ is an input alphabet,
$\delta : Q \times \Sigma \to Q$ is the transition function,
$q_0 \in Q$ is the initial state, and $F \subseteq Q$ is the set of
accepting states.
\end{definition}

\subsection{Duty Chains}
\begin{definition}
A \emph{duty chain} is a finite ordered sequence of transitions
\[
(q_0 \xrightarrow{a_1} q_1 \xrightarrow{a_2} q_2 \cdots
\xrightarrow{a_n} q_n)
\]
generated by repeated application of $\delta$.
\end{definition}

\subsection{Collapse Surfaces}
\begin{definition}
A \emph{collapse surface} is a subset $C \subseteq Q$ such that any
transition entering $C$ forces the automaton into a terminal or
degenerate state class.
\end{definition}

\subsection{Discrete Transition Operators}
\begin{definition}
A \emph{discrete transition operator} $T$ acts on state distributions
$\mu : Q \to [0,1]$ by
\[
(T\mu)(q) = \sum_{p \in Q} \mu(p) \cdot \mathbf{1}\big(q = \delta(p,a)\big)
\]
for fixed input symbol $a \in \Sigma$.
\end{definition}
