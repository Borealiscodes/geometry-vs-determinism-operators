\section{Cross--Module Consistency}

\subsection{Geometric--Spectral Consistency}

\begin{definition}
A system is \emph{geometric--spectral consistent} if geometric parallel
transport and spectral bundle transport agree on shared trajectories:
\[

= \mathcal{B}_\Lambda( P_{\gamma}(p) )
\]
for all curves $\gamma$ in $\M$.
\end{definition}

\begin{lemma}
If curvature is bounded and $\mathcal{B}_\Lambda$ varies smoothly, then
geometric--spectral consistency holds up to first order:
\[
\| T_{\gamma} - P_{\gamma} \|
\le C \cdot \mathrm{length}(\gamma).
\]
\end{lemma}

\begin{remark}
This condition ensures that geometric and spectral transport do not drift
apart during runtime execution.
\end{remark}
