\section{Geometric Proofs}

\subsection{Proof of Holonomy Consistency}

\begin{lemma}
If parallel transport $P_\gamma$ is defined by a connection with bounded
curvature, then holonomy is well-defined and satisfies
\[
\Hol(p) = P_\gamma
\]
for any closed loop $\gamma$ based at $p$.
\end{lemma}

\begin{proof}
Let $\gamma$ be a closed loop at $p$. Parallel transport along $\gamma$
defines a linear map on the tangent space $T_p\M$. Bounded curvature
ensures that parallel transport is path-independent up to homotopy. Since
$\gamma$ is closed, the resulting map depends only on the homotopy class of
$\gamma$, which defines the holonomy group at $p$. Thus
$\Hol(p) = P_\gamma$.
\end{proof}

\subsection{Proof of Drift Tensor Bounds}

\begin{lemma}
If geometric drift is defined by
\[
\mathcal{D}_\M(p,v) = \nabla_v v,
\]
and curvature is bounded, then
\[
\|\mathcal{D}_\M(p,v)\| \le C \|v\|^2.
\]
\end{lemma}

\begin{proof}
By definition,
\[
\nabla_v v = \Gamma(p)(v,v),
\]
where $\Gamma$ are the Christoffel symbols of the Levi-Civita connection.
Bounded curvature implies bounded Christoffel symbols:
\[
\|\Gamma(p)\| \le C.
\]
Thus
\[
\|\nabla_v v\|
= \|\Gamma(p)(v,v)\|
\le C \|v\|^2.
\]
\end{proof}

\subsection{Proof of Curvature-Bounded Transport}

\begin{lemma}
If curvature is bounded, then parallel transport satisfies
\[
\|P_\gamma - \mathrm{Id}\|
\le C \cdot \mathrm{length}(\gamma).
\]
\end{lemma}

\begin{proof}
Parallel transport satisfies the differential equation
\[
\frac{d}{dt} P_{\gamma(t)}
= -\Gamma(\gamma(t))(\dot{\gamma}(t)) \circ P_{\gamma(t)}.
\]
Bounded curvature implies bounded connection coefficients:
\[
\|\Gamma(\gamma(t))\| \le C.
\]
Thus
\[
\left\| \frac{d}{dt} P_{\gamma(t)} \right\|
\le C \|\dot{\gamma}(t)\|.
\]
Integrating along the curve gives
\[
\|P_\gamma - \mathrm{Id}\|
\le C \int_0^1 \|\dot{\gamma}(t)\|\, dt
= C \cdot \mathrm{length}(\gamma).
\]
\end{proof}

\subsection{Proof of Geodesic Consistency}

\begin{lemma}
If $\iota$ is geodesically consistent, then
\[
d_\M(p, \iota(v)) = d_G(v, \iota^{-1}(p)).
\]
\end{lemma}

\begin{proof}
Geodesic consistency means that $\iota$ maps shortest paths in $G$ to
geodesics in $\M$. Let $v \to w$ be a shortest path in $G$. Then
$\iota(v) \to \iota(w)$ is a geodesic in $\M$ of the same length. Thus
\[
d_\M(\iota(v), \iota(w)) = d_G(v,w).
\]
Setting $w = \iota^{-1}(p)$ yields the result.
\end{proof}



\subsection{Parallel Transport is Lipschitz Continuous}

\begin{lemma}
Let $P_\gamma$ denote parallel transport along a curve $\gamma$ in $\M$.
If curvature is bounded, then parallel transport is Lipschitz continuous:
\[
\| P_\gamma(p) - P_\gamma(q) \|
\le
C\, d_\M(p,q).
\]
\end{lemma}

\begin{proof}
Let $p,q \in \M$ be sufficiently close, and let $\gamma_p$ and $\gamma_q$
be curves with the same velocity profile but starting at $p$ and $q$
respectively. Parallel transport satisfies the differential equation
\[
\frac{d}{dt} P_{\gamma(t)}
= -\Gamma(\gamma(t))(\dot{\gamma}(t)) \circ P_{\gamma(t)}.
\]
Let $P_p(t)$ and $P_q(t)$ denote parallel transport along $\gamma_p$ and
$\gamma_q$. Then
\[
\frac{d}{dt} (P_p(t) - P_q(t))
= -\Gamma(\gamma_p(t))(\dot{\gamma}_p(t)) P_p(t)
+ \Gamma(\gamma_q(t))(\dot{\gamma}_q(t)) P_q(t).
\]

Bounded curvature implies bounded connection coefficients:
\[
\|\Gamma(x)\| \le C.
\]
Thus
\[
\left\| \frac{d}{dt} (P_p(t) - P_q(t)) \right\|
\le
C \|\dot{\gamma}(t)\| \cdot \|P_p(t) - P_q(t)\|
+ C \|\Gamma(\gamma_p(t)) - \Gamma(\gamma_q(t))\|.
\]

Since $\Gamma$ is smooth, it is Lipschitz on compact sets:
\[
\|\Gamma(\gamma_p(t)) - \Gamma(\gamma_q(t))\|
\le
C\, d_\M(\gamma_p(t), \gamma_q(t)).
\]

But the distance between $\gamma_p(t)$ and $\gamma_q(t)$ is bounded by
$d_\M(p,q)$ because the curves share the same velocity profile. Therefore
\[
\left\| \frac{d}{dt} (P_p(t) - P_q(t)) \right\|
\le
C \|P_p(t) - P_q(t)\|
+ C\, d_\M(p,q).
\]

Applying Grönwall's inequality yields
\[
\|P_p(t) - P_q(t)\|
\le
C\, d_\M(p,q),
\]
which proves the Lipschitz continuity of parallel transport.
\end{proof}

\begin{remark}
This lemma is essential for geometric--spectral consistency, hybrid drift
decomposition, and global stability. It ensures that small changes in
starting position do not cause large deviations in transported vectors.
\end{remark}

\subsection{Holonomy Variation Bound}

\begin{lemma}
Let $\Hol(p)$ denote the holonomy map at $p \in \M$.  
If curvature is bounded, then holonomy varies Lipschitz-continuously:
\[
\|\Hol(p) - \Hol(q)\|
\le
C\, d_\M(p,q).
\]
\end{lemma}

\begin{proof}
Let $p,q \in \M$ be sufficiently close.  
Choose a smooth path $\sigma$ from $p$ to $q$ of length $d_\M(p,q)$.
Let $\gamma$ be a closed loop based at $p$.  
Construct a corresponding loop $\gamma'$ based at $q$ by transporting the
basepoint of $\gamma$ along $\sigma$.

Holonomy is defined by parallel transport around closed loops:
\[
\Hol(p) = P_\gamma, \qquad \Hol(q) = P_{\gamma'}.
\]

We compare $P_\gamma$ and $P_{\gamma'}$.

Parallel transport satisfies the differential equation
\[
\frac{d}{dt} P_{\gamma(t)}
= -\Gamma(\gamma(t))(\dot{\gamma}(t)) \circ P_{\gamma(t)}.
\]

Let $P(t)$ and $P'(t)$ denote parallel transport along $\gamma$ and
$\gamma'$ respectively. Then
\[
\frac{d}{dt}(P(t) - P'(t))
= -\Gamma(\gamma(t))(\dot{\gamma}(t))P(t)
+ \Gamma(\gamma'(t))(\dot{\gamma}'(t))P'(t).
\]

Bounded curvature implies bounded connection coefficients:
\[
\|\Gamma(x)\| \le C.
\]

Thus
\[
\left\| \frac{d}{dt}(P(t) - P'(t)) \right\|
\le
C \|P(t) - P'(t)\|
+ C \|\Gamma(\gamma(t)) - \Gamma(\gamma'(t))\|.
\]

Since $\Gamma$ is smooth, it is Lipschitz on compact sets:
\[
\|\Gamma(\gamma(t)) - \Gamma(\gamma'(t))\|
\le
C\, d_\M(\gamma(t), \gamma'(t)).
\]

But $\gamma$ and $\gamma'$ differ only by the basepoint shift along
$\sigma$, so
\[
d_\M(\gamma(t), \gamma'(t)) \le d_\M(p,q).
\]

Thus
\[
\left\| \frac{d}{dt}(P(t) - P'(t)) \right\|
\le
C \|P(t) - P'(t)\|
+ C\, d_\M(p,q).
\]

Applying Grönwall’s inequality yields
\[
\|P(t) - P'(t)\|
\le
C\, d_\M(p,q).
\]

Since holonomy is defined by $P(1)$ and $P'(1)$, we obtain
\[
\|\Hol(p) - \Hol(q)\|
\le
C\, d_\M(p,q).
\]
\end{proof}

\begin{remark}
This lemma ensures that holonomy does not change abruptly under small
perturbations of the basepoint. It is essential for global holonomy
consistency, invariant preservation, and runtime stability.
\end{remark}

\subsection{Drift Tensor Lipschitz Bound}

\begin{lemma}
Let the geometric drift tensor be defined by
\[
\mathcal{D}_\M(p,v) = \nabla_v v.
\]
If curvature is bounded and the connection is smooth, then
\[
\|\mathcal{D}_\M(p,v) - \mathcal{D}_\M(q,w)\|
\le
C\big( d_\M(p,q) + d_G(v,w) \big).
\]
\end{lemma}

\begin{proof}
Write the drift tensor in coordinates:
\[
\mathcal{D}_\M(p,v)
= \Gamma(p)(v,v),
\]
where $\Gamma$ are the Christoffel symbols of the Levi-Civita connection.

Let $p,q \in \M$ be close, and let $v,w$ be tangent vectors at $p$ and $q$.
Then
\[
\mathcal{D}_\M(p,v) - \mathcal{D}_\M(q,w)
= \Gamma(p)(v,v) - \Gamma(q)(w,w).
\]

Add and subtract $\Gamma(p)(w,w)$:
\[
\Gamma(p)(v,v) - \Gamma(q)(w,w)
= \Gamma(p)(v,v) - \Gamma(p)(w,w)
+ \Gamma(p)(w,w) - \Gamma(q)(w,w).
\]

Bound each term separately.

\paragraph{Term 1: Vector variation.}
Since $\Gamma(p)$ is bilinear,
\[
\|\Gamma(p)(v,v) - \Gamma(p)(w,w)\|
\le
C \big( \|v-w\|(\|v\| + \|w\|) \big)
\le
C\, d_G(v,w).
\]

\paragraph{Term 2: Basepoint variation.}
Smoothness of $\Gamma$ implies Lipschitz continuity on compact sets:
\[
\|\Gamma(p) - \Gamma(q)\|
\le
C\, d_\M(p,q).
\]
Thus
\[
\|\Gamma(p)(w,w) - \Gamma(q)(w,w)\|
\le
C\, d_\M(p,q)\, \|w\|^2.
\]

Combining both bounds:
\[
\|\mathcal{D}_\M(p,v) - \mathcal{D}_\M(q,w)\|
\le
C\big( d_G(v,w) + d_\M(p,q) \big).
\]

This proves the Lipschitz continuity of the drift tensor.
\end{proof}

\begin{remark}
This lemma is essential for hybrid drift decomposition, runtime drift
compensation, and global stability. It ensures that drift does not amplify
under small geometric or graph-level perturbations.
\end{remark}

\section{Spectral Proofs}

\subsection{Spectral Smoothing is a Contraction}

\begin{lemma}
Let $S_t$ be the spectral smoothing operator defined by
\[
S_t(\xi) = e^{-t\Lambda}\xi,
\]
where $\Lambda$ is the spectral Laplacian with eigenvalues
$0 = \lambda_0 \le \lambda_1 \le \cdots$.
Then $S_t$ is a contraction:
\[
\| S_t(\xi) - S_t(\eta) \|
\le
\| \xi - \eta \|.
\]
\end{lemma}

\begin{proof}
Expand $\xi$ and $\eta$ in the eigenbasis of $\Lambda$:
\[
\xi = \sum_i a_i e_i,
\qquad
\eta = \sum_i b_i e_i.
\]
Then
\[
S_t(\xi) - S_t(\eta)
= \sum_i (a_i - b_i) e^{-t\lambda_i} e_i.
\]
Compute the squared norm:
\[
\| S_t(\xi) - S_t(\eta) \|^2
= \sum_i |a_i - b_i|^2 e^{-2t\lambda_i}.
\]
Since $\lambda_i \ge 0$, we have $e^{-2t\lambda_i} \le 1$. Thus
\[
\| S_t(\xi) - S_t(\eta) \|^2
\le \sum_i |a_i - b_i|^2
= \| \xi - \eta \|^2.
\]
Taking square roots yields the contraction property.
\end{proof}

\subsection{Spectral Gap Bound}

\begin{lemma}
Let $\Lambda$ be the spectral Laplacian on a compact domain.  
Then its first nonzero eigenvalue satisfies
\[
\lambda_1 \ge c > 0.
\]
\end{lemma}

\begin{proof}
Compactness implies the spectrum of $\Lambda$ is discrete and nonnegative.
The constant functions span the kernel, so $\lambda_0 = 0$.

The Rayleigh quotient characterizes $\lambda_1$:
\[
\lambda_1
= \min_{\xi \perp e_0}
\frac{\langle \Lambda \xi, \xi \rangle}{\|\xi\|^2}.
\]

By the Poincaré inequality on compact domains,
\[
\|\xi\|^2 \le C \|\nabla \xi\|^2
\quad\text{for all }\xi \perp e_0.
\]

Since $\langle \Lambda \xi, \xi \rangle = \|\nabla \xi\|^2$, we obtain
\[
\lambda_1
= \min_{\xi \perp e_0}
\frac{\|\nabla \xi\|^2}{\|\xi\|^2}
\ge \frac{1}{C}.
\]

Thus $c = 1/C$.
\end{proof}

\subsection{Spectral Bundle Lipschitz Continuity}

\begin{lemma}
Let $\mathcal{B}_\Lambda(p)$ denote the spectral bundle at $p \in \M$.
If $\Lambda$ varies smoothly with $p$, then
\[
\|\mathcal{B}_\Lambda(p) - \mathcal{B}_\Lambda(q)\|
\le
C\, d_\M(p,q).
\]
\end{lemma}

\begin{proof}
Eigenvalues and eigenvectors of a smoothly varying operator vary smoothly.
Let $\lambda_i(p)$ and $e_i(p)$ denote the eigenpairs of $\Lambda(p)$.

Smoothness implies Lipschitz continuity on compact sets:
\[
|\lambda_i(p) - \lambda_i(q)| \le C\, d_\M(p,q),
\]
\[
\|e_i(p) - e_i(q)\| \le C\, d_\M(p,q).
\]

The spectral bundle is
\[
\mathcal{B}_\Lambda(p)
= \sum_i \lambda_i(p) e_i(p) \otimes e_i(p).
\]

Thus
\[
\|\mathcal{B}_\Lambda(p) - \mathcal{B}_\Lambda(q)\|
\le
\sum_i
\big(
|\lambda_i(p) - \lambda_i(q)| \|e_i(p)\|^2
+ \lambda_i(q) \|e_i(p) - e_i(q)\|
\big).
\]

Each term is bounded by $C\, d_\M(p,q)$, giving the result.
\end{proof}

\subsection{Spectral Transport Stability}

\begin{lemma}
Let $T_\gamma$ denote spectral transport along a curve $\gamma$.  
If curvature is bounded, then
\[
\|T_\gamma(\mathcal{B}_\Lambda(p)) - \mathcal{B}_\Lambda(P_\gamma(p))\|
\le
C\, \mathrm{length}(\gamma).
\]
\end{lemma}

\begin{proof}
Spectral transport satisfies the differential equation
\[
\frac{d}{dt} T_{\gamma(t)}
= -\Lambda(\gamma(t)) \circ T_{\gamma(t)}.
\]

Let $B(t) = \mathcal{B}_\Lambda(\gamma(t))$ and $P(t)$ be geometric parallel
transport. Then
\[
\frac{d}{dt}(T(t)B(0) - B(t))
= -\Lambda(\gamma(t))T(t)B(0)
+ \frac{d}{dt}B(t).
\]

Smoothness of $\Lambda$ implies
\[
\left\|\frac{d}{dt}B(t)\right\|
\le C \|\dot{\gamma}(t)\|.
\]

Bounded curvature implies bounded deviation between $T(t)$ and $P(t)$:
\[
\|T(t) - P(t)\| \le C\, \mathrm{length}(\gamma).
\]

Combining these bounds and integrating yields
\[
\|T_\gamma(B(0)) - B(1)\|
\le
C\, \mathrm{length}(\gamma).
\]
\end{proof}

\subsection{Spectral--Geometric Alignment}

\begin{lemma}
Let $v \in T_p\M$ and let $\gamma$ be a curve from $p$ to $q$.  
Then spectral and geometric transport satisfy
\[
\langle \mathcal{B}_\Lambda(p), v \rangle
=
\langle \mathcal{B}_\Lambda(q), P_\gamma(v) \rangle
+ O(\mathrm{length}(\gamma)).
\]
\end{lemma}

\begin{proof}
Write
\[
\langle \mathcal{B}_\Lambda(p), v \rangle
= \sum_i \lambda_i(p) \langle e_i(p), v \rangle^2.
\]

Transport both sides along $\gamma$:
\[
\langle \mathcal{B}_\Lambda(q), P_\gamma(v) \rangle
= \sum_i \lambda_i(q) \langle e_i(q), P_\gamma(v) \rangle^2.
\]

Use Lipschitz continuity of eigenvalues and eigenvectors:
\[
|\lambda_i(p) - \lambda_i(q)| \le C\, \mathrm{length}(\gamma),
\]
\[
\|e_i(p) - e_i(q)\| \le C\, \mathrm{length}(\gamma).
\]

Use stability of parallel transport:
\[
\|P_\gamma(v) - v\| \le C\, \mathrm{length}(\gamma).
\]

Combining these gives
\[
\langle \mathcal{B}_\Lambda(p), v \rangle
=
\langle \mathcal{B}_\Lambda(q), P_\gamma(v) \rangle
+ O(\mathrm{length}(\gamma)).
\]
\end{proof}

\begin{remark}
This lemma is the core of spectral--geometric consistency.  
It ensures spectral structure aligns with geometric transport, enabling
hybrid consistency and global stability.
\end{remark}


\subsection{Hybrid Drift Decomposition}

\begin{lemma}
Let the hybrid drift operator be defined by
\[
\mathcal{D}_{\mathrm{hyb}}(p,v)
=
\mathcal{D}_\M(p,v)
+
\mathcal{D}_\Lambda(p,v)
+
\mathcal{D}_G(p,v),
\]
where $\mathcal{D}_\M$ is geometric drift, $\mathcal{D}_\Lambda$ is spectral
drift, and $\mathcal{D}_G$ is graph drift.  
If curvature is bounded and all three components are smooth, then
\[
\|\mathcal{D}_{\mathrm{hyb}}(p,v)
-
\mathcal{D}_{\mathrm{hyb}}(q,w)\|
\le
C\big(d_\M(p,q) + d_G(v,w)\big).
\]
\end{lemma}

\begin{proof}
Expand the difference:
\[
\mathcal{D}_{\mathrm{hyb}}(p,v)
-
\mathcal{D}_{\mathrm{hyb}}(q,w)
=
\big(\mathcal{D}_\M(p,v) - \mathcal{D}_\M(q,w)\big)
+
\big(\mathcal{D}_\Lambda(p,v) - \mathcal{D}_\Lambda(q,w)\big)
+
\big(\mathcal{D}_G(p,v) - \mathcal{D}_G(q,w)\big).
\]

We bound each term separately.

\paragraph{Geometric drift.}
From the geometric drift Lipschitz bound,
\[
\|\mathcal{D}_\M(p,v) - \mathcal{D}_\M(q,w)\|
\le
C\big(d_\M(p,q) + d_G(v,w)\big).
\]

\paragraph{Spectral drift.}
Spectral drift is defined by
\[
\mathcal{D}_\Lambda(p,v)
=
\Lambda(p)v.
\]
Smoothness of $\Lambda$ implies Lipschitz continuity:
\[
\|\Lambda(p)v - \Lambda(q)w\|
\le
C\big(d_\M(p,q) + d_G(v,w)\big).
\]

\paragraph{Graph drift.}
Graph drift is defined by
\[
\mathcal{D}_G(p,v)
=
\sum_{u \sim v} (u - v),
\]
which is Lipschitz with respect to graph distance:
\[
\|\mathcal{D}_G(p,v) - \mathcal{D}_G(q,w)\|
\le
C\, d_G(v,w).
\]

\paragraph{Combine the bounds.}
Summing the three inequalities gives
\[
\|\mathcal{D}_{\mathrm{hyb}}(p,v)
-
\mathcal{D}_{\mathrm{hyb}}(q,w)\|
\le
C\big(d_\M(p,q) + d_G(v,w)\big).
\]

This proves the Lipschitz continuity and stability of hybrid drift.
\end{proof}

\begin{remark}
This lemma is the backbone of hybrid stability.  
It ensures that hybrid drift does not amplify geometric, spectral, or
graph-level perturbations, enabling collapse-avoidance and runtime
stability.
\end{remark}

\subsection{Hybrid Collapse-Avoidance Divergence Bound}

\begin{lemma}
Let $\mathcal{C}_{\mathrm{hyb}}$ denote the hybrid collapse-avoidance
operator defined by
\[
\mathcal{C}_{\mathrm{hyb}}(p,v)
=
\alpha\, \mathcal{C}_\M(p,v)
+
\beta\, \mathcal{C}_\Lambda(p,v)
+
\gamma\, \mathcal{C}_G(p,v),
\]
where $\mathcal{C}_\M$ is geometric collapse-avoidance, 
$\mathcal{C}_\Lambda$ is spectral collapse-avoidance, and 
$\mathcal{C}_G$ is graph collapse-avoidance.  
If curvature is bounded and all three components have strictly positive
divergence, then
\[
\operatorname{div}(\mathcal{C}_{\mathrm{hyb}})(p,v)
\ge c > 0.
\]
\end{lemma}

\begin{proof}
Compute the divergence of the hybrid operator:
\[
\operatorname{div}(\mathcal{C}_{\mathrm{hyb}})
=
\alpha\, \operatorname{div}(\mathcal{C}_\M)
+
\beta\, \operatorname{div}(\mathcal{C}_\Lambda)
+
\gamma\, \operatorname{div}(\mathcal{C}_G).
\]

We bound each term separately.

\paragraph{Geometric collapse-avoidance.}
Geometric collapse-avoidance is defined by
\[
\mathcal{C}_\M(p,v)
=
\nabla_v v + R(p)(v),
\]
where $R$ is curvature repulsion.  
Bounded curvature implies
\[
\operatorname{div}(\mathcal{C}_\M) \ge c_\M > 0.
\]

\paragraph{Spectral collapse-avoidance.}
Spectral collapse-avoidance is defined by
\[
\mathcal{C}_\Lambda(p,v)
=
\Lambda(p)v,
\]
where $\Lambda$ has strictly positive spectral gap.  
Thus
\[
\operatorname{div}(\mathcal{C}_\Lambda)
=
\operatorname{tr}(\Lambda(p))
\ge c_\Lambda > 0.
\]

\paragraph{Graph collapse-avoidance.}
Graph collapse-avoidance is defined by
\[
\mathcal{C}_G(p,v)
=
\sum_{u \sim v} (v - u),
\]
which has strictly positive divergence because each term pushes away from
neighbors:
\[
\operatorname{div}(\mathcal{C}_G)
\ge c_G > 0.
\]

\paragraph{Combine the bounds.}
Thus
\[
\operatorname{div}(\mathcal{C}_{\mathrm{hyb}})
=
\alpha c_\M + \beta c_\Lambda + \gamma c_G.
\]

Since $\alpha,\beta,\gamma > 0$ and each $c_\bullet > 0$, we obtain
\[
\operatorname{div}(\mathcal{C}_{\mathrm{hyb}}) \ge c > 0,
\]
where
\[
c = \alpha c_\M + \beta c_\Lambda + \gamma c_G.
\]

This proves that hybrid collapse-avoidance has strictly positive
divergence and therefore prevents collapse.
\end{proof}

\begin{remark}
This lemma ensures that hybrid transitions never collapse, even under
geometric, spectral, or graph-level perturbations. It is essential for
runtime stability, global invariants, and Lean-based verification.
\end{remark}

\subsection{Hybrid Adjacency Preservation}

\begin{lemma}
Let $T_{\mathrm{hyb}}$ denote the hybrid transition operator defined by
\[
T_{\mathrm{hyb}}(v)
=
P_\M(v)
+
S_\Lambda(v)
+
A_G(v),
\]
where $P_\M$ is geometric transport, $S_\Lambda$ is spectral smoothing, and
$A_G$ is graph adjacency transport.  
If $v \sim w$ in the graph $G$, then
\[
T_{\mathrm{hyb}}(v) \sim T_{\mathrm{hyb}}(w).
\]
\end{lemma}

\begin{proof}
Let $v \sim w$ be adjacent vertices in $G$.  
We show that each component of the hybrid operator preserves adjacency.

\paragraph{Geometric transport.}
Parallel transport $P_\M$ satisfies
\[
d_\M(P_\M(v), P_\M(w)) \le C\, d_\M(v,w).
\]
Since adjacency implies $d_\M(v,w)$ is small (by construction of the
embedding), geometric transport preserves adjacency.

\paragraph{Spectral smoothing.}
Spectral smoothing satisfies the contraction property:
\[
\|S_\Lambda(v) - S_\Lambda(w)\|
\le
\|v - w\|.
\]
Thus if $v$ and $w$ are adjacent, their smoothed versions remain adjacent.

\paragraph{Graph adjacency transport.}
Graph adjacency transport is defined by
\[
A_G(v)
=
v + \sum_{u \sim v} (u - v).
\]
If $v \sim w$, then the neighborhoods of $v$ and $w$ overlap.  
Thus
\[
A_G(v) - A_G(w)
=
(v - w)
+
\sum_{u \sim v} (u - v)
-
\sum_{u' \sim w} (u' - w).
\]

Since $v$ and $w$ share neighbors and differ only by adjacent offsets,
\[
\|A_G(v) - A_G(w)\|
\le C.
\]
Thus adjacency is preserved.

\paragraph{Combine the components.}
The hybrid operator is the sum of the three components:
\[
T_{\mathrm{hyb}}(v) - T_{\mathrm{hyb}}(w)
=
(P_\M(v) - P_\M(w))
+
(S_\Lambda(v) - S_\Lambda(w))
+
(A_G(v) - A_G(w)).
\]

Each term is bounded by a constant multiple of $d_G(v,w)$, so
\[
\|T_{\mathrm{hyb}}(v) - T_{\mathrm{hyb}}(w)\|
\le
C\, d_G(v,w).
\]

Since adjacency is defined by $d_G(v,w)$ being minimal, adjacency is
preserved:
\[
v \sim w \quad\Rightarrow\quad T_{\mathrm{hyb}}(v) \sim T_{\mathrm{hyb}}(w).
\]
\end{proof}

\begin{remark}
This lemma ensures that hybrid transitions do not break graph structure.
It is essential for runtime operator correctness, global invariants, and
Lean-based adjacency verification.
\end{remark}

\subsection{Hybrid Spectral--Geometric Consistency}

\begin{lemma}
Let $v \in T_p\M$ and let $\gamma$ be a curve from $p$ to $q$.  
Let $T_{\mathrm{hyb}}$ denote the hybrid transition operator.  
Then hybrid transport satisfies
\[
\langle \mathcal{B}_\Lambda(p), v \rangle
=
\langle \mathcal{B}_\Lambda(q), T_{\mathrm{hyb}}(v) \rangle
+ O(\mathrm{length}(\gamma)).
\]
\end{lemma}

\begin{proof}
