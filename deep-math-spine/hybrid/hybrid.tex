\section{Hybrid Geometric–Discrete Operators}

\subsection{Manifold–Graph Embeddings}
\begin{definition}
A \emph{manifold–graph embedding} is a pair
\[
(\iota, G)
\]
where $\iota : V(G) \to \M$ maps graph vertices into a smooth manifold
$\M$, and edges correspond to geodesic or near-geodesic relations under
$\iota$.
\end{definition}

\subsection{Hybrid Transition Operators}
\begin{definition}
Given a manifold $\M$ and a discrete state graph $G$, a
\emph{hybrid transition operator} $H$ acts on pairs $(p, q)$ where
$p \in \M$ and $q \in V(G)$, producing
\[
H(p,q) = \big( \Phi(p,q),\; \delta(q,a) \big)
\]
for some geometric update rule $\Phi$ and discrete transition $\delta$.
\end{definition}

\subsection{Spectral–Duty Hybrid Operators}
\begin{definition}
A \emph{spectral–duty hybrid operator} combines spectral propagation on
$\M$ with discrete duty-chain evolution on $G$, producing updates of the
form
\[
S(p,q) = \big( e^{-t\Delta} p,\; T\mu(q) \big)
\]
where $e^{-t\Delta}$ is the heat semigroup and $T$ is a discrete
transition operator.
\end{definition}
